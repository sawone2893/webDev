4. Conditionals:Adding elements on the basis of conditions.
    Example1:Using ternary operator
        export default function Product({title,description,features}){
                    return (
                        <div className="Product:>
                        <h1>{title}</h1>
                        <p> {description}<p>
                        <p> {features.map((feature)=><li>{feature}</li>)}<p>
                        {price>=3000?<p>Discount of 5%</p>:null}
                        </div>
                    );
                }
    Example2: Using And Operator
        export default function Product({title,description,features}){
